import React, { useState, useEffect } from 'react';
import { timeEntriesAPI } from '../services/api';
import { Check, X, Eye, Filter } from 'lucide-react';

const Approvals = () => {
  const [pendingEntries, setPendingEntries] = useState([]);
  const [loading, setLoading] = useState(true);
  const [selectedEntry, setSelectedEntry] = useState(null);
  const [showRejectModal, setShowRejectModal] = useState(false);
  const [rejectionReason, setRejectionReason] = useState('');
  const [filter, setFilter] = useState('PENDING');

  useEffect(() => {
    fetchPendingApprovals();
  }, []);

  const fetchPendingApprovals = async () => {
    setLoading(true);
    try {
      const response = await timeEntriesAPI.getPendingApprovals();
      setPendingEntries(response.data.data || []);
    } catch (error) {
      console.error('Error fetching pending approvals:', error);
    } finally {
      setLoading(false);
    }
  };

  const handleApprove = async (entryId) => {
    try {
      await timeEntriesAPI.approve(entryId, { status: 'APPROVED' });
      fetchPendingApprovals();
    } catch (error) {
      console.error('Error approving entry:', error);
      alert('Failed to approve entry');
    }
  };

  const handleReject = async () => {
    if (!rejectionReason.trim()) {
      alert('Please provide a reason for rejection');
      return;
    }
    try {
      await timeEntriesAPI.approve(selectedEntry.id, {
        status: 'REJECTED',
        rejection_reason: rejectionReason,
      });
      setShowRejectModal(false);
      setSelectedEntry(null);
      setRejectionReason('');
      fetchPendingApprovals();
    } catch (error) {
      console.error('Error rejecting entry:', error);
      alert('Failed to reject entry');
    }
  };

  const getStatusBadge = (status) => `badge badge-${status.toLowerCase()}`;

  if (loading) {
    return (
      <div className="loading-container">
        <div className="spinner"></div>
      </div>
    );
  }

  return (
    <div>
      <header className="header">
        <div className="header-title">
          <h2>Approvals</h2>
          <p>Review and approve time entries</p>
        </div>
        <div className="header-actions">
          <select
            className="form-select"
            style={{ width: 'auto' }}
            value={filter}
            onChange={(e) => setFilter(e.target.value)}
          >
            <option value="PENDING">Pending</option>
            <option value="APPROVED">Approved</option>
            <option value="REJECTED">Rejected</option>
            <option value="ALL">All</option>
          </select>
        </div>
      </header>

      <div className="page-content">
        {pendingEntries.length === 0 ? (
          <div className="card">
            <div className="empty-state">
              <div className="empty-state-icon">
                <Check size={48} />
              </div>
              <div className="empty-state-title">All caught up!</div>
              <div className="empty-state-text">No pending approvals at this time.</div>
            </div>
          </div>
        ) : (
          <div className="card">
            <div className="card-header">
              <h3 className="card-title">Pending Approvals ({pendingEntries.length})</h3>
            </div>
            <div className="table-container">
              <table className="table">
                <thead>
                  <tr>
                    <th>Employee</th>
                    <th>Project</th>
                    <th>Date</th>
                    <th>Hours</th>
                    <th>Notes</th>
                    <th>Actions</th>
                  </tr>
                </thead>
                <tbody>
                  {pendingEntries.map((entry) => (
                    <tr key={entry.id}>
                      <td>
                        <div className="flex items-center gap-2">
                          <div className="avatar avatar-sm">
                            {entry.user_name.split(' ').map(n => n[0]).join('')}
                          </div>
                          <span>{entry.user_name}</span>
                        </div>
                      </td>
                      <td>{entry.project_name}</td>
                      <td>{entry.work_date}</td>
                      <td>{entry.hours}h</td>
                      <td>{entry.notes || '-'}</td>
                      <td>
                        <div className="flex gap-2">
                          <button
                            className="btn btn-sm btn-success"
                            onClick={() => handleApprove(entry.id)}
                            title="Approve"
                          >
                            <Check size={16} />
                          </button>
                          <button
                            className="btn btn-sm btn-danger"
                            onClick={() => { setSelectedEntry(entry); setShowRejectModal(true); }}
                            title="Reject"
                          >
                            <X size={16} />
                          </button>
                        </div>
                      </td>
                    </tr>
                  ))}
                </tbody>
              </table>
            </div>
          </div>
        )}

        {/* Quick Actions */}
        <div className="stats-grid" style={{ marginTop: '24px' }}>
          <div className="stat-card">
            <div className="stat-card-header">
              <div className="stat-icon orange">
                <Eye size={24} />
              </div>
            </div>
            <div className="stat-value">{pendingEntries.length}</div>
            <div className="stat-label">Pending Approvals</div>
          </div>
          <div className="stat-card">
            <div className="stat-card-header">
              <div className="stat-icon teal">
                <Filter size={24} />
              </div>
            </div>
            <div className="stat-value">
              {pendingEntries.reduce((sum, e) => sum + parseFloat(e.hours || 0), 0).toFixed(1)}h
            </div>
            <div className="stat-label">Total Hours Pending</div>
          </div>
        </div>
      </div>

      {/* Reject Modal */}
      {showRejectModal && (
        <div className="modal-overlay" onClick={() => setShowRejectModal(false)}>
          <div className="modal" onClick={(e) => e.stopPropagation()}>
            <div className="modal-header">
              <h3 className="modal-title">Reject Time Entry</h3>
              <button className="btn btn-icon" onClick={() => setShowRejectModal(false)}>
                <X size={20} />
              </button>
            </div>
            <div className="modal-body">
              <div className="form-group">
                <label className="form-label">Employee</label>
                <input type="text" className="form-input" value={selectedEntry?.user_name} disabled />
              </div>
              <div className="form-group">
                <label className="form-label">Project</label>
                <input type="text" className="form-input" value={selectedEntry?.project_name} disabled />
              </div>
              <div className="form-group">
                <label className="form-label">Hours</label>
                <input type="text" className="form-input" value={`${selectedEntry?.hours}h`} disabled />
              </div>
              <div className="form-group">
                <label className="form-label">Reason for Rejection</label>
                <textarea
                  className="form-input"
                  placeholder="Please provide a reason for rejecting this time entry..."
                  value={rejectionReason}
                  onChange={(e) => setRejectionReason(e.target.value)}
                  rows={4}
                  required
                />
              </div>
            </div>
            <div className="modal-footer">
              <button className="btn btn-secondary" onClick={() => setShowRejectModal(false)}>
                Cancel
              </button>
              <button className="btn btn-danger" onClick={handleReject}>
                Reject Entry
              </button>
            </div>
          </div>
        </div>
      )}
    </div>
  );
};

export default Approvals;

import React, { useState, useEffect } from 'react';
import useAuthStore from '../stores/authStore';
import { timeEntriesAPI, reportsAPI, projectsAPI } from '../services/api';
import {
  Clock,
  CheckCircle,
  XCircle,
  AlertCircle,
  TrendingUp,
  Calendar,
  Users,
  FolderKanban,
} from 'lucide-react';

const Dashboard = () => {
  const { user, isAdmin, isManager } = useAuthStore();
  const [stats, setStats] = useState({
    totalHours: 0,
    pendingEntries: 0,
    approvedEntries: 0,
    rejectedEntries: 0,
    totalProjects: 0,
    totalUsers: 0,
  });
  const [recentEntries, setRecentEntries] = useState([]);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    fetchDashboardData();
  }, []);

  const fetchDashboardData = async () => {
    try {
      const today = new Date();
      const startOfMonth = new Date(today.getFullYear(), today.getMonth(), 1);

      // Fetch time entries
      const entriesResponse = await timeEntriesAPI.list({
        start_date: startOfMonth.toISOString().split('T')[0],
        end_date: today.toISOString().split('T')[0],
      });

      const entries = entriesResponse.data.data.entries || [];
      
      // Calculate stats
      const totalHours = entries.reduce((sum, e) => sum + parseFloat(e.hours || 0), 0);
      const pendingEntries = entries.filter((e) => e.status === 'PENDING').length;
      const approvedEntries = entries.filter((e) => e.status === 'APPROVED').length;
      const rejectedEntries = entries.filter((e) => e.status === 'REJECTED').length;

      // Fetch projects
      const projectsResponse = await projectsAPI.list();
      const totalProjects = projectsResponse.data.data.total || 0;

      setStats({
        totalHours,
        pendingEntries,
        approvedEntries,
        rejectedEntries,
        totalProjects,
        totalUsers: 0,
      });

      setRecentEntries(entries.slice(0, 5));
    } catch (error) {
      console.error('Error fetching dashboard data:', error);
    } finally {
      setLoading(false);
    }
  };

  const getGreeting = () => {
    const hour = new Date().getHours();
    if (hour < 12) return 'Good morning';
    if (hour < 17) return 'Good afternoon';
    return 'Good evening';
  };

  const getStatusBadge = (status) => {
    return `badge badge-${status.toLowerCase()}`;
  };

  if (loading) {
    return (
      <div className="loading-container">
        <div className="spinner"></div>
      </div>
    );
  }

  return (
    <div>
      <header className="header">
        <div className="header-title">
          <h2>{getGreeting()}, {user?.name}!</h2>
          <p>Here's what's happening with your time tracking</p>
        </div>
      </header>

      <div className="page-content">
        <div className="stats-grid">
          <div className="stat-card">
            <div className="stat-card-header">
              <div className="stat-icon teal">
                <Clock size={24} />
              </div>
            </div>
            <div className="stat-value">{stats.totalHours.toFixed(1)}h</div>
            <div className="stat-label">Total Hours This Month</div>
          </div>

          <div className="stat-card">
            <div className="stat-card-header">
              <div className="stat-icon orange">
                <AlertCircle size={24} />
              </div>
            </div>
            <div className="stat-value">{stats.pendingEntries}</div>
            <div className="stat-label">Pending Approvals</div>
          </div>

          <div className="stat-card">
            <div className="stat-card-header">
              <div className="stat-icon green">
                <CheckCircle size={24} />
              </div>
            </div>
            <div className="stat-value">{stats.approvedEntries}</div>
            <div className="stat-label">Approved Entries</div>
          </div>

          <div className="stat-card">
            <div className="stat-card-header">
              <div className="stat-icon blue">
                <FolderKanban size={24} />
              </div>
            </div>
            <div className="stat-value">{stats.totalProjects}</div>
            <div className="stat-label">Active Projects</div>
          </div>
        </div>

        <div className="card" style={{ marginBottom: '24px' }}>
          <div className="card-header">
            <h3 className="card-title">Recent Time Entries</h3>
          </div>
          <div className="table-container">
            {recentEntries.length === 0 ? (
              <div className="empty-state">
                <div className="empty-state-icon">
                  <Calendar size={48} />
                </div>
                <div className="empty-state-title">No time entries yet</div>
                <div className="empty-state-text">Start tracking your time by adding entries in the Timesheets section.</div>
              </div>
            ) : (
              <table className="table">
                <thead>
                  <tr>
                    <th>Project</th>
                    <th>Date</th>
                    <th>Hours</th>
                    <th>Status</th>
                    <th>Notes</th>
                  </tr>
                </thead>
                <tbody>
                  {recentEntries.map((entry) => (
                    <tr key={entry.id}>
                      <td>{entry.project_name}</td>
                      <td>{entry.work_date}</td>
                      <td>{entry.hours}h</td>
                      <td>
                        <span className={getStatusBadge(entry.status)}>
                          {entry.status}
                        </span>
                      </td>
                      <td>{entry.notes || '-'}</td>
                    </tr>
                  ))}
                </tbody>
              </table>
            )}
          </div>
        </div>

        {isAdmin() && (
          <div className="card">
            <div className="card-header">
              <h3 className="card-title">Quick Actions</h3>
            </div>
            <div className="card-body">

                  <FolderKanban size={18} />
                  Manage Projects
                </button>
              </div>

        )}
      </div>

import React, { useState } from 'react';
import { useNavigate, Link } from 'react-router-dom';
import { authAPI } from '../services/api';
import { Mail, ArrowLeft } from 'lucide-react';

const ForgotPassword = () => {
  const navigate = useNavigate();
  const [email, setEmail] = useState('');
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState('');
  const [success, setSuccess] = useState(false);

  const handleSubmit = async (e) => {
    e.preventDefault();
    setLoading(true);
    setError('');
    setSuccess(false);

    try {
      const response = await authAPI.requestPasswordReset(email);
      if (response.status === 200 || response.status === 201) {
        setSuccess(true);
        setEmail('');
      }
    } catch (err) {
      setError(err.response?.data?.detail || 'Failed to send reset link. Please try again.');
    } finally {
      setLoading(false);
    }
  };

  return (
    <div className="login-container">
      <div className="login-card">
        <div className="login-header">
          <div style={{ marginBottom: '16px' }}>
            <Mail size={32} color="var(--primary)" />
          </div>
          <h2 style={{ marginBottom: '8px' }}>Reset Password</h2>
          <div className="login-subtitle">Enter your email to receive a password reset link</div>
        </div>

        {error && (
          <div className="login-error" style={{ marginBottom: '20px' }}>
            {error}
          </div>
        )}

        {success ? (
          <div style={{
            padding: '16px',
            borderRadius: '8px',
            background: 'rgba(34, 197, 94, 0.1)',
            border: '1px solid rgba(34, 197, 94, 0.3)',
            marginBottom: '20px',
            textAlign: 'center'
          }}>
            <div style={{ color: 'var(--success)', fontWeight: 500, marginBottom: '8px' }}>
              ✓ Reset link sent!
            </div>
            <div style={{ fontSize: '14px', color: 'var(--text-secondary)', marginBottom: '16px' }}>
              Check your email for a password reset link. If you don't see it, check your spam folder.
            </div>
            <button
              onClick={() => navigate('/login')}
              className="btn btn-primary btn-block"
              style={{ width: '100%' }}
            >
              Back to Login
            </button>
          </div>
        ) : (
          <form className="login-form" onSubmit={handleSubmit}>
            <div className="form-group">
              <label className="form-label">Email Address</label>
              <input
                type="email"
                className="form-input"
                placeholder="Enter your email"
                value={email}
                onChange={(e) => setEmail(e.target.value)}
                required
              />
            </div>

            <button
              type="submit"
              className="btn btn-primary btn-block"
              disabled={loading}
              style={{ width: '100%', marginTop: '10px' }}
            >
              {loading ? 'Sending...' : 'Send Reset Link'}
            </button>
          </form>
        )}

        <div style={{ marginTop: '24px', textAlign: 'center' }}>
          <Link
            to="/login"
            style={{
              color: 'var(--primary)',
              fontWeight: 500,
              display: 'flex',
              alignItems: 'center',
              justifyContent: 'center',
              gap: '8px'
            }}
          >
            <ArrowLeft size={16} />
            Back to Login
          </Link>
        </div>
      </div>
    </div>
  );
};

export default ForgotPassword;

import React, { useState } from 'react';
import { useNavigate, useLocation } from 'react-router-dom';
import useAuthStore from '../stores/authStore';
import {
  LayoutDashboard,
  Calendar,
  CheckSquare,
  FolderKanban,
  BarChart3,
  Users,
  LogOut,
  ChevronLeft,
  ChevronRight,
} from 'lucide-react';

const Layout = ({ children }) => {
  const navigate = useNavigate();
  const location = useLocation();
  const { user, logout, isAdmin, isManager } = useAuthStore();
  const [collapsed, setCollapsed] = useState(false);

  const navItems = [
    { path: '/dashboard', icon: LayoutDashboard, label: 'Dashboard' },
    { path: '/timesheets', icon: Calendar, label: 'Timesheets' },
    ...(isAdmin() || isManager()
      ? [{ path: '/approvals', icon: CheckSquare, label: 'Approvals' }]
      : []),
    { path: '/projects', icon: FolderKanban, label: 'Projects' },
    { path: '/reports', icon: BarChart3, label: 'Reports' },
    ...(isAdmin()
      ? [{ path: '/users', icon: Users, label: 'User Provisioning' }]
      : []),
  ];

  const handleLogout = () => {
    logout();
    navigate('/');
  };

  const getInitials = (name) => {
    return name
      ?.split(' ')
      .map((n) => n[0])
      .join('')
      .toUpperCase()
      .slice(0, 2) || 'U';
  };

  const getRoleBadgeClass = (role) => {
    switch (role) {
      case 'Admin': return 'role-badge admin';
      case 'Manager': return 'role-badge manager';
      default: return 'role-badge member';
    }
  };

  return (
    <div className="app-container">
      <aside
        className="sidebar"
        style={{ width: collapsed ? '80px' : '260px', transition: 'width 0.3s ease' }}
      >
        <div className="sidebar-logo" style={{ padding: '20px', textAlign: 'center' }}>
          <img
            src="/tickmarks-logo.png"
            alt="Tickmarks Logo"
            style={{
              height: collapsed ? '24px' : '40px',
              maxWidth: '100%',
              transition: 'height 0.3s ease'
            }}
          />
          {!collapsed && (
            <div style={{ color: 'white', fontWeight: 'bold', marginTop: '8px', fontSize: '18px' }}>
              Timesheet
            </div>
          )}
        </div>

        <nav className="sidebar-nav">
          {navItems.map((item) => {
            const Icon = item.icon;
            const isActive = location.pathname === item.path;
            return (
              <div
                key={item.path}
                className={`nav-item ${isActive ? 'active' : ''}`}
                onClick={() => navigate(item.path)}
                title={collapsed ? item.label : ''}
              >
                <Icon size={20} />
                {!collapsed && <span>{item.label}</span>}
              </div>
            );
          })}
        </nav>

        <div className="sidebar-footer">
          <div className="user-info" style={{ flexDirection: collapsed ? 'column' : 'row', alignItems: 'center' }}>
            <div className="user-avatar">{getInitials(user?.name)}</div>
            {!collapsed && (
              <div className="user-details">
                <div className="user-name">{user?.name}</div>
                <div className="user-role">
                  {user?.roles?.map((r) => (
                    <span key={r.id || r.value} className={getRoleBadgeClass(r.value)}>
                      {r.value}
                    </span>
                  ))}
                </div>
              </div>
            )}
          </div>
          {!collapsed && (
            <button
              className="btn btn-secondary"
              style={{
                width: '100%',
                marginTop: '16px',
                background: 'rgba(255,255,255,0.1)',
                color: 'white',
                borderColor: 'rgba(255,255,255,0.2)'
              }}
              onClick={handleLogout}
            >
              <LogOut size={18} />
              Sign Out
            </button>
          )}
        </div>

        <button
          className="btn-icon"
          style={{
            position: 'absolute',
            right: -12,
            top: '50%',
            transform: 'translateY(-50%)',
            background: 'white',
            boxShadow: '0 2px 8px rgba(0,0,0,0.15)',
            borderRadius: '50%',
            width: '24px',
            height: '24px',
            padding: 0,
            display: 'flex',
            alignItems: 'center',
            justifyContent: 'center',
            zIndex: 100
          }}
          onClick={() => setCollapsed(!collapsed)}
        >
          {collapsed ? <ChevronRight size={16} /> : <ChevronLeft size={16} />}
        </button>
      </aside>

      <main className="main-content">{children}</main>
    </div>
  );
};

export default Layout;

      console.log('Projects response:', response);
      
      // Handle different response structures
      const projectsList = response?.data?.data?.projects || response?.data?.data || response?.data || [];
      let projectsArray = Array.isArray(projectsList) ? projectsList : [];
      
      // Ensure it's always an array
      if (!Array.isArray(projectsArray)) {
        projectsArray = [];
      }
      
      // Fetch members for each project
      const projectsWithMembers = await Promise.all(
        projectsArray.map(async (project) => {
          try {
            const membersRes = await projectsAPI.getMembers(project.id);
            return {
              ...project,
              members: Array.isArray(membersRes?.data?.data) ? membersRes.data.data : []
            };
          } catch (err) {
            console.error(`Error fetching members for project ${project.id}:`, err);
            return { ...project, members: [] };
          }
        })
      );
      
      setProjects(Array.isArray(projectsWithMembers) ? projectsWithMembers : []);

      console.error('Error fetching projects:', error.message, error.response?.data);
      setError(`Failed to load projects: ${error.response?.data?.message || error.message}`);
      setProjects([]);
    } finally {
      setLoading(false);
    }

  const fetchInitialData = async () => {
    try {
      const [mgrRes, userRes] = await Promise.all([usersAPI.getManagers(), usersAPI.list()]);

      let usersList = userRes?.data?.data || [];

          usersList = usersList.users;

          const values = Object.values(usersList);
          if (Array.isArray(values[0]) && values[0].length > 0 && values[0][0]?.id) {
            usersList = values[0];
          } else {
            usersList = values.filter(v => Array.isArray(v) && v[0]?.id);
            if (usersList.length > 0) {
              usersList = usersList[0];
            }
          }
        }

import React, { useState } from 'react';
import { useNavigate, useSearchParams, Link } from 'react-router-dom';
import { authAPI } from '../services/api';
import { Eye, EyeOff, Lock } from 'lucide-react';

const ResetPassword = () => {
  const navigate = useNavigate();
  const [searchParams] = useSearchParams();
  const token = searchParams.get('token');

  const [formData, setFormData] = useState({
    password: '',
    confirmPassword: '',
  });
  const [showPassword, setShowPassword] = useState(false);
  const [showConfirmPassword, setShowConfirmPassword] = useState(false);
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState('');
  const [success, setSuccess] = useState(false);

  const handleSubmit = async (e) => {
    e.preventDefault();
    setError('');

    if (!token) {
      setError('Invalid or missing reset token. Please request a new password reset.');
      return;
    }

    if (formData.password.length < 6) {
      setError('Password must be at least 6 characters long.');
      return;
    }

    if (formData.password !== formData.confirmPassword) {
      setError('Passwords do not match.');
      return;
    }

    setLoading(true);

    try {
      await authAPI.resetPassword(token, formData.password);
      setSuccess(true);
      setFormData({ password: '', confirmPassword: '' });
    } catch (err) {
      setError(err.response?.data?.detail || 'Failed to reset password. The link may have expired.');
    } finally {
      setLoading(false);
    }
  };

  if (!token) {
    return (
      <div className="login-container">
        <div className="login-card">
          <div className="login-header">
            <div style={{ marginBottom: '16px' }}>
              <Lock size={32} color="var(--error)" />
            </div>
            <h2 style={{ marginBottom: '8px' }}>Invalid Reset Link</h2>
          </div>
          <div className="login-error" style={{ marginBottom: '20px' }}>
            No reset token found. Please request a new password reset.
          </div>
          <Link to="/forgot-password" className="btn btn-primary btn-block" style={{ width: '100%' }}>
            Request New Reset Link
          </Link>
        </div>
      </div>
    );
  }

  return (
    <div className="login-container">
      <div className="login-card">
        <div className="login-header">
          <div style={{ marginBottom: '16px' }}>
            <Lock size={32} color="var(--primary)" />
          </div>
          <h2 style={{ marginBottom: '8px' }}>Create New Password</h2>
          <div className="login-subtitle">Enter your new password below</div>
        </div>

        {error && (
          <div className="login-error" style={{ marginBottom: '20px' }}>
            {error}
          </div>
        )}

        {success ? (
          <div style={{
            padding: '16px',
            borderRadius: '8px',
            background: 'rgba(34, 197, 94, 0.1)',
            border: '1px solid rgba(34, 197, 94, 0.3)',
            marginBottom: '20px',
            textAlign: 'center'
          }}>
            <div style={{ color: 'var(--success)', fontWeight: 500, marginBottom: '8px' }}>
              ✓ Password Reset Successful!
            </div>
            <div style={{ fontSize: '14px', color: 'var(--text-secondary)', marginBottom: '16px' }}>
              Your password has been changed. You can now log in with your new password.
            </div>
            <button
              onClick={() => navigate('/login')}
              className="btn btn-primary btn-block"
              style={{ width: '100%' }}
            >
              Go to Login
            </button>
          </div>
        ) : (
          <form className="login-form" onSubmit={handleSubmit}>
            <div className="form-group">
              <label className="form-label">New Password</label>
              <div style={{ position: 'relative' }}>
                <input
                  type={showPassword ? 'text' : 'password'}

    const sunday = new Date(monday);
    sunday.setDate(monday.getDate() + 6);
    sunday.setHours(23, 59, 59, 999);

    const checkDate = new Date(date);
    return checkDate >= monday && checkDate <= sunday;

const ENABLE_SUPABASE_FALLBACK = process.env.REACT_APP_ENABLE_SUPABASE_FALLBACK === 'true';
const API_TIMEOUT = 5000; // 5 seconds timeout before trying Supabase

// Helper to add timeout to promises
const withTimeout = (promise, timeoutMs) => {
  return Promise.race([
    promise,
    new Promise((_, reject) =>
      setTimeout(() => reject(new Error('Request timeout')), timeoutMs)
    )
  ]);
};

// Fallback API wrapper
export const apiWithFallback = {
  // Time Entries
  async getTimeEntries(filters) {
    if (!ENABLE_SUPABASE_FALLBACK) {
      return api.get('/time-entries', { params: filters });
    }

    try {
      // Try backend first with timeout
      return await withTimeout(
        api.get('/time-entries', { params: filters }),
        API_TIMEOUT
      );
    } catch (error) {
      console.warn('Backend timeout/error, trying Supabase:', error.message);
      try {
        const data = await supabaseService.getTimeEntries(filters);
        return { data: { data: { entries: data } } };
      } catch (supabaseError) {
        console.error('Supabase fallback failed:', supabaseError);

  },

// Direct Supabase connection (fallback when backend is down)
// NOTE: This requires @supabase/supabase-js client library
// Install with: npm install @supabase/supabase-js

const SUPABASE_URL = process.env.REACT_APP_SUPABASE_URL;
const SUPABASE_ANON_KEY = process.env.REACT_APP_SUPABASE_ANON_KEY;

let supabase = null;

// Lazy load Supabase client only when needed
const getSupabaseClient = async () => {
  if (supabase) return supabase;
  
  if (!SUPABASE_URL || !SUPABASE_ANON_KEY) {
    console.warn('Supabase credentials not configured');
    return null;
  }

  try {
    const { createClient } = await import('@supabase/supabase-js');
    supabase = createClient(SUPABASE_URL, SUPABASE_ANON_KEY);
    return supabase;
  } catch (error) {
    console.error('Failed to initialize Supabase client:', error);
    return null;
  }
};

export const supabaseService = {
  // Get the authenticated Supabase client
  async getClient() {
    return getSupabaseClient();
  },

  // Get current user session
  async getSession() {
    const client = await getSupabaseClient();
    if (!client) return null;
    
    const token = localStorage.getItem('token');
    if (!token) return null;
    
    return { token };
  },

  // Query time entries
  async getTimeEntries(filters) {
    const client = await getSupabaseClient();
    if (!client) throw new Error('Supabase client not available');

    const token = localStorage.getItem('token');
    if (!token) throw new Error('Not authenticated');

    let query = client
      .from('time_entries')
      .select('*');

    if (filters?.start_date) {
      query = query.gte('work_date', filters.start_date);
    }
    if (filters?.end_date) {
      query = query.lte('work_date', filters.end_date);
    }
    if (filters?.user_id) {
      query = query.eq('user_id', filters.user_id);
    }

    const { data, error } = await query;
    if (error) throw error;
    return data;
  },

  // Create time entry
  async createTimeEntry(data) {
    const client = await getSupabaseClient();
    if (!client) throw new Error('Supabase client not available');

    const token = localStorage.getItem('token');
    if (!token) throw new Error('Not authenticated');

    const { data: entry, error } = await client
      .from('time_entries')
      .insert([{ ...data, status: 'PENDING' }])
      .select();

    if (error) throw error;
    return entry;
  },

  // Get projects
  async getProjects() {
    const client = await getSupabaseClient();
    if (!client) throw new Error('Supabase client not available');

    const token = localStorage.getItem('token');
    if (!token) throw new Error('Not authenticated');

    const { data, error } = await client
      .from('projects')
      .select('*');

    if (error) throw error;
    return data;
  },

  // Get users
  async getUsers() {
    const client = await getSupabaseClient();
    if (!client) throw new Error('Supabase client not available');

    const token = localStorage.getItem('token');
    if (!token) throw new Error('Not authenticated');

    const { data, error } = await client
      .from('users')
      .select('*');

    if (error) throw error;
    return data;
  },

  // Get reports
  async getReports(filters) {
    const client = await getSupabaseClient();
    if (!client) throw new Error('Supabase client not available');

    const token = localStorage.getItem('token');
    if (!token) throw new Error('Not authenticated');

    let query = client
      .from('time_entries')
      .select('*')
      .eq('status', 'APPROVED');

    if (filters?.start_date) {
      query = query.gte('work_date', filters.start_date);
    }
    if (filters?.end_date) {
      query = query.lte('work_date', filters.end_date);
    }

    const { data, error } = await query;
    if (error) throw error;
    return data;
  },
};

export default supabaseService;

import { create } from 'zustand';
import { authAPI } from '../services/api';

const useAuthStore = create((set, get) => ({
  user: JSON.parse(localStorage.getItem('user')) || null,
  token: localStorage.getItem('token') || null,
  isAuthenticated: !!localStorage.getItem('token'),
  loading: false,
  error: null,

  login: async (email, password) => {
    set({ loading: true, error: null });
    try {
      const response = await authAPI.login(email, password);
      const { data } = response.data;
      
      if (response.data.status === 'error') {
        set({ loading: false, error: response.data.message });
        return false;
      }

      localStorage.setItem('token', data.token);
      localStorage.setItem('user', JSON.stringify(data.profile));
      
      set({
        user: data.profile,
        token: data.token,
        isAuthenticated: true,
        loading: false,
        error: null,
      });
      
      return true;
    } catch (error) {
      const message = error.response?.data?.message || 'Login failed';
      set({ loading: false, error: message });
      return false;
    }
  },

  register: async (userData) => {
    set({ loading: true, error: null });
    try {
      const response = await authAPI.register(userData);
      
      if (response.data.status === 'error') {
        set({ loading: false, error: response.data.message });
        return false;
      }

      set({ loading: false });
      return true;
    } catch (error) {
      const message = error.response?.data?.message || 'Registration failed';
      set({ loading: false, error: message });
      return false;
    }
  },

  logout: () => {
    localStorage.removeItem('token');
    localStorage.removeItem('user');
    set({
      user: null,
      token: null,
      isAuthenticated: false,
      error: null,
    });
  },

  fetchCurrentUser: async () => {
    try {
      const response = await authAPI.getCurrentUser();
      const user = response.data.data;
      localStorage.setItem('user', JSON.stringify(user));
      set({ user });
    } catch (error) {
      console.error('Failed to fetch current user:', error);
    }
  },

  hasRole: (roleName) => {
    const user = get().user;
    if (!user || !user.roles) return false;
    return user.roles.some(role => role.value === roleName);
  },

  isAdmin: () => get().hasRole('Admin'),
  isManager: () => get().hasRole('Manager'),
  isMember: () => get().hasRole('Member'),

  clearError: () => set({ error: null }),
}));

export default useAuthStore;

import React, { useState, useEffect } from 'react';
import { BrowserRouter, Routes, Route, Navigate } from 'react-router-dom';
import useAuthStore from './stores/authStore';
import Login from './components/Login';
import Register from './components/Register';
import ForgotPassword from './components/ForgotPassword';
import ResetPassword from './components/ResetPassword';
import Dashboard from './components/Dashboard';
import Timesheets from './components/Timesheets';
import Approvals from './components/Approvals';
import Projects from './components/Projects';
import Reports from './components/Reports';
import UserProvisioning from './components/UserProvisioning';
import Layout from './components/Layout';

function App() {
  const { isAuthenticated, fetchCurrentUser, token } = useAuthStore();
  const [initializing, setInitializing] = useState(true);

  useEffect(() => {
    const init = async () => {
      if (token) {
        await fetchCurrentUser();
      }
      setInitializing(false);
    };
    init();
  }, [token, fetchCurrentUser]);

  if (initializing) {
    return (
      <div className="loading-container">
        <div className="spinner"></div>
      </div>
    );
  }

  return (
    <BrowserRouter>
      <Routes>
        <Route
          path="/"
          element={
            isAuthenticated ? (
              <Navigate to="/dashboard" replace />
            ) : (
              <Login />
            )
          }
        />
        <Route path="/register" element={<Register />} />
        <Route path="/forgot-password" element={<ForgotPassword />} />
        <Route path="/reset-password" element={<ResetPassword />} />
        <Route
          path="/*"
          element={
            isAuthenticated ? (
              <Layout>
                <Routes>
                  <Route path="/dashboard" element={<Dashboard />} />
                  <Route path="/timesheets" element={<Timesheets />} />
                  <Route path="/approvals" element={<Approvals />} />
                  <Route path="/projects" element={<Projects />} />
                  <Route path="/reports" element={<Reports />} />
                  <Route path="/users" element={<UserProvisioning />} />
                </Routes>
              </Layout>
            ) : (
              <Navigate to="/" replace />
            )
          }
        />
      </Routes>
    </BrowserRouter>
  );
}

export default App;

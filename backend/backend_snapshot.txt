async def get_current_user(
    credentials: HTTPAuthorizationCredentials = Depends(security),
    db: Session = Depends(get_db)
) -> User:
    token = credentials.credentials
    payload = decode_access_token(token)
    if not payload:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Invalid or expired token"
        )
    
    user_id = payload.get("sub")
    if not user_id:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Invalid token payload"
        )
    
    user = db.query(User).filter(User.id == int(user_id)).first()
    if not user:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="User not found"
        )
    
    return user


def get_user_profile(user: User, db: Session) -> dict:
    """Build user profile response with roles."""
    manager_name = None
    if user.parent_id:
        manager = db.query(User).filter(User.id == user.parent_id).first()
        if manager:
            manager_name = manager.name
    
    roles = []
    for role in user.roles:
        roles.append({"id": str(role.id), "value": role.name})
    
    return {
        "id": user.id,
        "name": user.name,
        "email": user.email,
        "status": user.status,
        "parent_id": user.parent_id,
        "manager_name": manager_name,
        "created_at": user.created_at,
        "roles": roles
    }


@router.post("/login", response_model=LoginResponse)
async def login(request: LoginRequest, db: Session = Depends(get_db)):
    try:
        user = db.query(User).filter(User.email == request.email).first()
        
        if not user or not verify_password(request.password, user.password_hash):
            return LoginResponse(
                status="error",
                statusCode=401,
                message="Invalid email or password",
                data=None,
                errors=["Invalid email or password"]
            )
        
        if not user.status:
            return LoginResponse(
                status="error",
                statusCode=403,
                message="Account is deactivated",
                data=None,
                errors=["Account is deactivated"]
            )
        
        if not user.email_verified:
            return LoginResponse(
                status="error",
                statusCode=403,
                message="Please verify your email address",
                data=None,
                errors=["Email not verified"]
            )
        
        access_token = create_access_token(data={"sub": str(user.id)})
        profile = get_user_profile(user, db)
        
        return LoginResponse(
            status="success",
            statusCode=200,
            message="Login successful",
            data=TokenResponse(token=access_token, profile=UserProfileResponse(**profile)),
            errors=[]
        )
    except Exception as e:
        import traceback
        print(f"Login error: {str(e)}")
        traceback.print_exc()
        return LoginResponse(
            status="error",
            statusCode=500,

        )

        name=request.name,
        email=request.email,
        password_hash=hash_password(request.password),
        status=True,
        email_verified=False,  # SOC2 Compliance: must click email link
        verification_token=secrets.token_urlsafe(32)
    )
    db.add(user)
    db.commit()
    db.refresh(user)
    
    # Assign roles
    if request.roles:
        for role_name in request.roles:
            role = db.query(Role).filter(Role.name == role_name).first()
            if role:
                user_role = UserRole(user_id=user.id, role_id=role.id)
                db.add(user_role)
    else:
        # Default role is Member
        member_role = db.query(Role).filter(Role.name == "Member").first()
        if member_role:
            user_role = UserRole(user_id=user.id, role_id=member_role.id)
            db.add(user_role)
    
    db.commit()
    
    # Send verification email
    try:
        await email_service.send_verification_email(
            email=user.email,
            name=user.name,
            token=user.verification_token
        )
    except Exception as e:
        # Log error but don't crash registration; user can resend later
        print(f"Email error: {str(e)}")
    
    return LoginResponse(
        status="success",
        statusCode=201,
        message="Registration successful. Please check your email to verify your account.",
        data=None,
        errors=[]
    )

@router.get("/summary")
async def get_report_summary(
    start_date: Optional[date] = None,
    end_date: Optional[date] = None,
    project_id: Optional[int] = None,
    user_id: Optional[int] = None,
    reporting_area: Optional[str] = None,
    current_user: User = Depends(get_current_user),
    db: Session = Depends(get_db)
):
    """Generate time entry summary report."""
    
    # Default to current month if no dates provided
    if not start_date:
        start_date = date.today().replace(day=1)
    if not end_date:
        end_date = date.today()
    
    query = db.query(TimeEntry).filter(
        TimeEntry.work_date >= start_date,
        TimeEntry.work_date <= end_date,
        TimeEntry.status == "APPROVED"
    )
    
    # Handle access control
    is_admin = check_admin(current_user)
    is_manager = check_manager(current_user)
    
    if is_admin:
        # Admin can see all entries
        pass
    elif is_manager:
        # Manager can see entries of their team members
        team_member_ids = db.query(User.id).filter(User.parent_id == current_user.id).all()
        team_member_ids = [m[0] for m in team_member_ids]
        if team_member_ids:
            query = query.filter(TimeEntry.user_id.in_(team_member_ids))
        else:
            # If no team members, show nothing
            query = query.filter(TimeEntry.user_id == -1)
    else:
        # Regular users see only their own entries
        query = query.filter(TimeEntry.user_id == current_user.id)
    
    if project_id:
        query = query.filter(TimeEntry.project_id == project_id)
    if user_id and is_admin:
        query = query.filter(TimeEntry.user_id == user_id)
    
    entries = query.all()
    
    # Group by project
    project_summary = {}
    for entry in entries:
        if entry.project_id not in project_summary:
            project = db.query(Project).filter(Project.id == entry.project_id).first()
            project_summary[entry.project_id] = {
                "project_id": entry.project_id,
                "project_name": project.name if project else "Unknown",
                "client": project.client if project else "Unknown",
                "reporting_area": project.reporting_area if project else None,
                "total_hours": Decimal(0),
                "user_hours": {}
            }
        
        project_summary[entry.project_id]["total_hours"] += entry.hours
        
        if entry.user_id not in project_summary[entry.project_id]["user_hours"]:
            user = db.query(User).filter(User.id == entry.user_id).first()
            project_summary[entry.project_id]["user_hours"][entry.user_id] = {
                "user_id": entry.user_id,
                "user_name": user.name if user else "Unknown",
                "hours": Decimal(0)
            }
        
        project_summary[entry.project_id]["user_hours"][entry.user_id]["hours"] += entry.hours
    
    # Convert user_hours to list
    result = []
    for pid, pdata in project_summary.items():
        if reporting_area and pdata["reporting_area"] != reporting_area:
            continue
        result.append({
            **pdata,
            "user_hours": list(pdata["user_hours"].values())
        })
    
    # Calculate totals
    total_hours = sum(float(p["total_hours"]) for p in result)
    
    return {
        "status": "success",
        "statusCode": 200,
        "message": "Report generated successfully",
        "data": {
            "start_date": start_date.isoformat(),
            "end_date": end_date.isoformat(),
            "total_hours": total_hours,
            "total_projects": len(result),
            "project_summary": result
        },
        "errors": []
    }

async def update_time_entry(
    entry_id: int,
    request: TimeEntryUpdate,
    current_user: User = Depends(get_current_user),
    db: Session = Depends(get_db)
):
    try:
        entry = db.query(TimeEntry).filter(TimeEntry.id == entry_id).first()
        if not entry:
            raise HTTPException(
                status_code=status.HTTP_404_NOT_FOUND,
                detail="Time entry not found"
            )
        
        # Only owner can update
        if entry.user_id != current_user.id:
            raise HTTPException(
                status_code=status.HTTP_403_FORBIDDEN,
                detail="Access denied"
            )
        
        # Allow updating PENDING entries and REJECTED entries (for resubmission)
        if entry.status not in ["PENDING", "REJECTED"]:
            raise HTTPException(
                status_code=status.HTTP_400_BAD_REQUEST,
                detail="Cannot modify approved entries. Only pending or rejected entries can be edited."
            )
        
        # If updating a rejected entry, prepare for resubmission
        is_resubmitting = entry.status == "REJECTED"
        
        # Validate hours if provided
        if request.hours is not None:
            try:
                hours_decimal = request.hours if isinstance(request.hours, type(1.0)) or hasattr(request.hours, '__float__') else float(request.hours)
                if hours_decimal < 0 or hours_decimal > 24:
                    raise HTTPException(
                        status_code=status.HTTP_400_BAD_REQUEST,
                        detail="Hours must be between 0 and 24"
                    )
            except (ValueError, TypeError) as e:
                raise HTTPException(
                    status_code=status.HTTP_400_BAD_REQUEST,
                    detail=f"Invalid hours value: {str(e)}"
                )
        
        if request.project_id is not None:
            # Verify project exists
            project = db.query(Project).filter(Project.id == request.project_id).first()
            if not project:
                raise HTTPException(
                    status_code=status.HTTP_400_BAD_REQUEST,
                    detail="Invalid project ID"
                )
            entry.project_id = request.project_id
        
        if request.work_date is not None:
            entry.work_date = request.work_date
        if request.hours is not None:
            entry.hours = request.hours
        if request.notes is not None:
            entry.notes = request.notes
        
        # If this is a resubmission, reset status to PENDING
        if is_resubmitting:
            entry.status = "PENDING"
            entry.rejection_reason = None
            entry.approved_by_id = None
            entry.approval_date = None
        
        db.add(entry)
        db.commit()
        db.refresh(entry)
        
        return {
            "status": "success",
            "statusCode": 200,
            "message": "Time entry updated successfully",
            "data": build_entry_response(entry, db),
            "errors": []
        }
    except HTTPException:
        raise
    except Exception as e:
        db.rollback()
        print(f"Error updating time entry: {str(e)}")
        import traceback
        traceback.print_exc()
        raise HTTPException(
            status_code=status.HTTP_500_INTERNAL_SERVER_ERROR,
            detail=f"Error updating time entry: {str(e)}"
        )

import os
from datetime import datetime, timedelta
from typing import Optional
from jose import jwt
from passlib.context import CryptContext

# Setup password hashing
pwd_context = CryptContext(schemes=["bcrypt"], deprecated="auto")

SECRET_KEY = os.getenv("JWT_SECRET_KEY", "7f3b89e2c1d4a0f6b8e2d4c1a0f6b8e2")
ALGORITHM = "HS256"

def hash_password(password: str) -> str:
    return pwd_context.hash(password)

def verify_password(plain_password: str, hashed_password: str) -> bool:
    return pwd_context.verify(plain_password, hashed_password)

def create_access_token(data: dict, expires_delta: Optional[timedelta] = None):
    to_encode = data.copy()
    expire = datetime.utcnow() + (expires_delta or timedelta(minutes=15))
    to_encode.update({"exp": expire})
    return jwt.encode(to_encode, SECRET_KEY, algorithm=ALGORITHM)

def decode_access_token(token: str):
    try:
        return jwt.decode(token, SECRET_KEY, algorithms=[ALGORITHM])
    except Exception:
        return None

class EmailService:
    def __init__(self):
        self.host = settings.SMTP_HOST
        self.port = settings.SMTP_PORT
        self.username = settings.SMTP_USER
        self.password = settings.SMTP_PASSWORD
        self.from_email = settings.SMTP_FROM
        self.frontend_url = settings.FRONTEND_URL

    async def send_email(
        self,
        to_email: str,
        subject: str,
        html_content: str,
        text_content: Optional[str] = None
    ) -> bool:
        """Send an email using SMTP."""
        if not self.username or not self.password:
            print(f"Email not configured. Would send to {to_email}: {subject}")
            return False

        message = MIMEMultipart("alternative")
        message["From"] = self.from_email
        message["To"] = to_email
        message["Subject"] = subject

        if text_content:
            message.attach(MIMEText(text_content, "plain"))
        message.attach(MIMEText(html_content, "html"))

        try:
            await aiosmtplib.send(
                message,
                hostname=self.host,
                port=self.port,
                username=self.username,
                password=self.password,
                start_tls=True
            )
            return True
        except Exception as e:
            print(f"Failed to send email: {e}")
            return False

from datetime import datetime, timedelta, date
from apscheduler.schedulers.asyncio import AsyncIOScheduler
from apscheduler.triggers.cron import CronTrigger
from sqlalchemy.orm import Session
from app.core.database import SessionLocal
from app.models.models import User, Project, TimeEntry
from app.services.email_service import email_service

scheduler = AsyncIOScheduler()


def get_week_date_range():
    """Get the start and end dates of the current week."""
    today = date.today()
    start = today - timedelta(days=today.weekday())  # Monday
    end = start + timedelta(days=6)  # Sunday
    return start, end


async def send_weekly_manager_reports():
    """Send weekly pending time entries report to all project managers."""
    db = SessionLocal()
    try:
        week_start, week_end = get_week_date_range()
        
        # Get all active projects with managers
        projects = db.query(Project).filter(Project.status == True).all()
        
        # Group entries by manager
        manager_entries = {}
        
        for project in projects:
            # Get pending entries for this project in the current week
            entries = db.query(TimeEntry).filter(
                TimeEntry.project_id == project.id,
                TimeEntry.status == "PENDING",
                TimeEntry.work_date >= week_start,
                TimeEntry.work_date <= week_end
            ).all()
            
            for entry in entries:
                user = db.query(User).filter(User.id == entry.user_id).first()
                
                entry_data = {
                    "user_name": user.name if user else "Unknown",
                    "project_name": project.name,
                    "work_date": entry.work_date.isoformat(),
                    "hours": float(entry.hours),
                    "notes": entry.notes or ""
                }
                
                if project.manager_id not in manager_entries:
                    manager = db.query(User).filter(User.id == project.manager_id).first()
                    manager_entries[project.manager_id] = {
                        "manager_email": manager.email if manager else "",
                        "manager_name": manager.name if manager else "",
                        "entries": []
                    }
                
                manager_entries[project.manager_id]["entries"].append(entry_data)
        
        # Send emails to each manager
        for manager_id, data in manager_entries.items():
            if data["manager_email"] and data["entries"]:
                await email_service.send_weekly_report(
                    manager_email=data["manager_email"],
                    manager_name=data["manager_name"],
                    pending_entries=data["entries"],
                    week_start=week_start.strftime("%Y-%m-%d"),
                    week_end=week_end.strftime("%Y-%m-%d")
                )
                
    finally:
        db.close()


def start_scheduler():
    """Start the APScheduler for scheduled tasks."""
    # Schedule weekly report every Sunday at 9 AM
    scheduler.add_job(
        send_weekly_manager_reports,
        CronTrigger(day_of_week="sun", hour=9, minute=0),
        id="weekly_manager_report",
        replace_existing=True
    )
    
    scheduler.start()
    print("Scheduler started. Weekly reports will be sent on Sundays at 9 AM.")


def stop_scheduler():
    """Stop the scheduler."""
    scheduler.shutdown()

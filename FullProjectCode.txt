export class ApiError extends Error {
  statusCode: number;
  constructor(message: string, statusCode: number) {
    super(message);

async function request<T>(endpoint: string, options: RequestInit = {}): Promise<T> {
  const token = useAuthStore.getState().token;
  const headers: HeadersInit = { 'Content-Type': 'application/json', ...options.headers };
  if (token) { (headers as Record<string, string>)['Authorization'] = `Bearer ${token}`; }
  
  const url = `${API_BASE}${endpoint}`;
  const response = await fetch(url, { ...options, headers });
  const data = await response.json();

  if (!response.ok) {
    throw new ApiError(data.detail || data.message || 'An error occurred', response.status);
  }
  return data;
}

import os

# Adds the backend directory to sys.path so app modules can be imported
sys.path.insert(0, os.path.dirname(os.path.dirname(os.path.abspath(__file__))))

from app.core.database import Base
from app.models import models
from app.core.config import settings

# This is the Alembic Config object
config = context.config

# Dynamically set the sqlalchemy.url from our app settings
if settings.DATABASE_URL:
    # Handles conversion if Render provides 'postgres://' instead of 'postgresql://'
    database_url = settings.DATABASE_URL
    if database_url.startswith("postgres://"):
        database_url = database_url.replace("postgres://", "postgresql://", 1)
    config.set_main_option("sqlalchemy.url", database_url)

# Interpret the config file for Python logging
if config.config_file_name is not None:
    fileConfig(config.config_file_name)

# Set the metadata for 'autogenerate' support
target_metadata = Base.metadata

def run_migrations_offline() -> None:
    """Run migrations in 'offline' mode."""
    url = config.get_main_option("sqlalchemy.url")
    context.configure(
        url=url,
        target_metadata=target_metadata,
        literal_binds=True,
        dialect_opts={"paramstyle": "named"},
    )

    with context.begin_transaction():
        context.run_migrations()

def run_migrations_online() -> None:
    """Run migrations in 'online' mode."""
    # Pulls the configuration from the [alembic] section of alembic.ini
    configuration = config.get_section(config.config_ini_section)
    
    connectable = engine_from_config(
        configuration,
        prefix="sqlalchemy.",
        poolclass=pool.NullPool,
    )

    with connectable.connect() as connection:
        context.configure(
            connection=connection, 
            target_metadata=target_metadata
        )

        with context.begin_transaction():
            context.run_migrations()

if context.is_offline_mode():
    run_migrations_offline()
else:
    run_migrations_online()

async def get_current_user(
    credentials: HTTPAuthorizationCredentials = Depends(security),
    db: Session = Depends(get_db)
) -> User:
    token = credentials.credentials
    payload = decode_access_token(token)
    if not payload:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Invalid or expired token"
        )
    
    user_id = payload.get("sub")
    if not user_id:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Invalid token payload"
        )
    
    user = db.query(User).filter(User.id == int(user_id)).first()
    if not user:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="User not found"
        )
    
    return user


def get_user_profile(user: User, db: Session) -> dict:
    """Build user profile response with roles."""
    manager_name = None
    if user.parent_id:
        manager = db.query(User).filter(User.id == user.parent_id).first()
        if manager:
            manager_name = manager.name
    
    roles = []
    for role in user.roles:
        roles.append({"id": str(role.id), "value": role.name})
    
    return {
        "id": user.id,
        "name": user.name,
        "email": user.email,
        "status": user.status,
        "parent_id": user.parent_id,
        "manager_name": manager_name,
        "created_at": user.created_at,

    existing_user = db.query(User).filter(User.email == request.email).first()
    if existing_user:
        return LoginResponse(
            status="error",
            statusCode=400,
            message="Email already registered",
            data=None,
            errors=["Email already registered"]
        )
    
    # Create user with fixed indentation
    user = User(
        name=request.name,
        email=request.email,
        password_hash=hash_password(request.password),
        status=True,
        email_verified=False,  # SOC2 Compliance: must click email link
        verification_token=secrets.token_urlsafe(32)
    )
    db.add(user)
    db.commit()
    db.refresh(user)
    
    # Assign roles
    if request.roles:
        for role_name in request.roles:
            role = db.query(Role).filter(Role.name == role_name).first()
            if role:
                user_role = UserRole(user_id=user.id, role_id=role.id)
                db.add(user_role)
    else:
        # Default role is Member
        member_role = db.query(Role).filter(Role.name == "Member").first()
        if member_role:
            user_role = UserRole(user_id=user.id, role_id=member_role.id)
            db.add(user_role)
    
    db.commit()
    
    # Send verification email
    try:
        await email_service.send_verification_email(
            email=user.email,
            name=user.name,
            token=user.verification_token
        )
    except Exception as e:
        # Log error but don't crash registration; user can resend later
        print(f"Email error: {str(e)}")
    
    return LoginResponse(
        status="success",
        statusCode=201,
        message="Registration successful. Please check your email to verify your account.",
        data=None,
        errors=[]
    )

async def get_report_summary(
    start_date: Optional[date] = None,
    end_date: Optional[date] = None,
    project_id: Optional[int] = None,
    user_id: Optional[int] = None,
    reporting_area: Optional[str] = None,
    current_user: User = Depends(get_current_user),
    db: Session = Depends(get_db)
):
    """Generate time entry summary report."""
    
    # Default to current month if no dates provided
    if not start_date:
        start_date = date.today().replace(day=1)
    if not end_date:
        end_date = date.today()
    
    query = db.query(TimeEntry).filter(
        TimeEntry.work_date >= start_date,
        TimeEntry.work_date <= end_date,
        TimeEntry.status == "APPROVED"
    )
    
    # Non-admin users can only see their own entries
    if not check_admin(current_user):
        query = query.filter(TimeEntry.user_id == current_user.id)
    
    if project_id:
        query = query.filter(TimeEntry.project_id == project_id)
    if user_id and check_admin(current_user):
        query = query.filter(TimeEntry.user_id == user_id)
    
    entries = query.all()
    
    # Group by project
    project_summary = {}
    for entry in entries:
        if entry.project_id not in project_summary:
            project = db.query(Project).filter(Project.id == entry.project_id).first()
            project_summary[entry.project_id] = {
                "project_id": entry.project_id,
                "project_name": project.name if project else "Unknown",
                "client": project.client if project else "Unknown",
                "reporting_area": project.reporting_area if project else None,
                "total_hours": Decimal(0),
                "user_hours": {}
            }
        
        project_summary[entry.project_id]["total_hours"] += entry.hours
        
        if entry.user_id not in project_summary[entry.project_id]["user_hours"]:
            user = db.query(User).filter(User.id == entry.user_id).first()
            project_summary[entry.project_id]["user_hours"][entry.user_id] = {
                "user_id": entry.user_id,
                "user_name": user.name if user else "Unknown",
                "hours": Decimal(0)
            }
        
        project_summary[entry.project_id]["user_hours"][entry.user_id]["hours"] += entry.hours
    
    # Convert user_hours to list
    result = []
    for pid, pdata in project_summary.items():
        if reporting_area and pdata["reporting_area"] != reporting_area:
            continue
        result.append({
            **pdata,
            "user_hours": list(pdata["user_hours"].values())
        })
    
    # Calculate totals
    total_hours = sum(float(p["total_hours"]) for p in result)
    
    return {
        "status": "success",
        "statusCode": 200,
        "message": "Report generated successfully",
        "data": {
            "start_date": start_date.isoformat(),
            "end_date": end_date.isoformat(),
            "total_hours": total_hours,
            "total_projects": len(result),
            "project_summary": result
        },
        "errors": []
    }

from app.core.database import get_db
from app.api.auth import get_current_user
from app.models.models import User, Project, ProjectMember, TimeEntry
from app.schemas.schemas import (
    TimeEntryCreate, TimeEntryUpdate, TimeEntryApproval, TimeEntryResubmit
)
from app.services.email_service import email_service

router = APIRouter(prefix="/time-entries", tags=["Time Entries"])


def check_admin(user: User) -> bool:
    return any(role.name == "Admin" for role in user.roles)


def check_manager(user: User) -> bool:
    return any(role.name == "Manager" for role in user.roles)


def build_entry_response(entry: TimeEntry, db: Session) -> dict:
    user = db.query(User).filter(User.id == entry.user_id).first()
    project = db.query(Project).filter(Project.id == entry.project_id).first()
    approver = None
    if entry.approved_by_id:
        approver = db.query(User).filter(User.id == entry.approved_by_id).first()
    
    return {
        "id": entry.id,
        "user_id": entry.user_id,
        "user_name": user.name if user else "Unknown",
        "project_id": entry.project_id,
        "project_name": project.name if project else "Unknown",
        "work_date": entry.work_date.isoformat(),
        "hours": float(entry.hours),
        "notes": entry.notes,
        "status": entry.status,
        "approved_by_id": entry.approved_by_id,
        "approved_by_name": approver.name if approver else None,
        "approval_date": entry.approval_date.isoformat() if entry.approval_date else None,
        "rejection_reason": entry.rejection_reason,
        "resubmission_notes": entry.resubmission_notes,
        "created_at": entry.created_at.isoformat()
    }


@router.get("")
async def list_time_entries(
    page: int = Query(1, ge=1),
    size: int = Query(100, ge=1, le=100),
    start_date: Optional[date] = None,
    end_date: Optional[date] = None,
    project_id: Optional[int] = None,
    status_filter: Optional[str] = None,
    current_user: User = Depends(get_current_user),
    db: Session = Depends(get_db)
):
    query = db.query(TimeEntry)
    
    # Non-admin users can only see their own entries
    if not check_admin(current_user):
        query = query.filter(TimeEntry.user_id == current_user.id)
    
    if start_date:
        query = query.filter(TimeEntry.work_date >= start_date)
    if end_date:
        query = query.filter(TimeEntry.work_date <= end_date)
    if project_id:
        query = query.filter(TimeEntry.project_id == project_id)
    if status_filter:
        query = query.filter(TimeEntry.status == status_filter)
    
    total = query.count()
    entries = query.order_by(TimeEntry.work_date.desc()).offset(
        (page - 1) * size
    ).limit(size).all()
    
    return {
        "status": "success",
        "statusCode": 200,
        "message": "Time entries retrieved successfully",
        "data": {
            "entries": [build_entry_response(e, db) for e in entries],
            "total": total,
            "page": page,
            "size": size
        },
        "errors": []
    }


@router.post("")
async def create_time_entry(
    request: TimeEntryCreate,
    current_user: User = Depends(get_current_user),
    db: Session = Depends(get_db)
):
    # Check if user is member of the project
    member = db.query(ProjectMember).filter(
        ProjectMember.project_id == request.project_id,
        ProjectMember.user_id == current_user.id
    ).first()
    
    if not member and not check_admin(current_user):
        raise HTTPException(
            status_code=status.HTTP_403_FORBIDDEN,
            detail="You are not assigned to this project"
        )
    
    # Check for duplicate entry
    existing = db.query(TimeEntry).filter(
        TimeEntry.user_id == current_user.id,
        TimeEntry.project_id == request.project_id,
        TimeEntry.work_date == request.work_date
    ).first()
    
    if existing:
        return {
            "status": "error",
            "statusCode": 400,
            "message": "Time entry already exists for this date and project",
            "data": None,
            "errors": ["Duplicate entry"]
        }
    
    entry = TimeEntry(
        user_id=current_user.id,
        project_id=request.project_id,
        work_date=request.work_date,
        hours=request.hours,
        notes=request.notes,
        status="PENDING"
    )
    db.add(entry)
    db.commit()
    db.refresh(entry)
    
    return {
        "status": "success",
        "statusCode": 201,
        "message": "Time entry created successfully",
        "data": build_entry_response(entry, db),
        "errors": []
    }

@router.post("/{entry_id}/approve")
async def approve_time_entry(
    entry_id: int,
    request: TimeEntryApproval,
    current_user: User = Depends(get_current_user),
    db: Session = Depends(get_db)
):
    entry = db.query(TimeEntry).filter(TimeEntry.id == entry_id).first()
    if not entry:
        raise HTTPException(
            status_code=status.HTTP_404_NOT_FOUND,
            detail="Time entry not found"
        )
    
    # Get project to check if user is manager
    project = db.query(Project).filter(Project.id == entry.project_id).first()
    if not project:
        raise HTTPException(
            status_code=status.HTTP_404_NOT_FOUND,
            detail="Project not found"
        )
    
    # Check if user can approve (admin or project manager)
    if not check_admin(current_user) and project.manager_id != current_user.id:
        raise HTTPException(
            status_code=status.HTTP_403_FORBIDDEN,
            detail="Only admin or project manager can approve entries"
        )
    
    if entry.status != "PENDING":
        raise HTTPException(
            status_code=status.HTTP_400_BAD_REQUEST,
            detail="Entry is not pending approval"
        )
    
    if request.status == "APPROVED":
        entry.status = "APPROVED"
        entry.approved_by_id = current_user.id
        entry.approval_date = datetime.utcnow()
    elif request.status == "REJECTED":
        entry.status = "REJECTED"
        entry.rejection_reason = request.rejection_reason
    else:
        raise HTTPException(
            status_code=status.HTTP_400_BAD_REQUEST,
            detail="Invalid status. Must be APPROVED or REJECTED"
        )
    
    db.commit()
    db.refresh(entry)
    
    # Send notification email
    user = db.query(User).filter(User.id == entry.user_id).first()
    if user:
        await email_service.send_approval_notification(
            employee_email=user.email,
            employee_name=user.name,
            entry_details=build_entry_response(entry, db),
            status=entry.status,
            rejection_reason=entry.rejection_reason
        )
    
    return {
        "status": "success",
        "statusCode": 200,
        "message": f"Time entry {entry.status.lower()}",
        "data": build_entry_response(entry, db),
        "errors": []
    }

@router.get("/pending/approvals")
async def get_pending_approvals(
    current_user: User = Depends(get_current_user),
    db: Session = Depends(get_db)
):
    """Get all pending entries for projects managed by current user."""
    if not check_admin(current_user) and not check_manager(current_user):
        raise HTTPException(
            status_code=status.HTTP_403_FORBIDDEN,
            detail="Manager access required"
        )
    
    # Get projects where user is manager
    managed_projects = db.query(Project).filter(
        Project.manager_id == current_user.id
    ).all()
    
    project_ids = [p.id for p in managed_projects]
    
    if check_admin(current_user):
        # Admin can see all pending entries
        entries = db.query(TimeEntry).filter(
            TimeEntry.status == "PENDING"
        ).all()
    else:
        # Manager can only see entries for their projects
        entries = db.query(TimeEntry).filter(
            TimeEntry.status == "PENDING",
            TimeEntry.project_id.in_(project_ids)
        ).all()
    
    return {
        "status": "success",
        "statusCode": 200,
        "message": "Pending approvals retrieved successfully",
        "data": [build_entry_response(e, db) for e in entries],
        "errors": []
    }

from datetime import datetime, timedelta
from typing import Optional
from jose import JWTError, jwt
from passlib.context import CryptContext
from app.core.config import settings

pwd_context = CryptContext(schemes=["bcrypt"], deprecated="auto")


def hash_password(password: str) -> str:
    return pwd_context.hash(password)


def verify_password(plain_password: str, hashed_password: str) -> bool:
    return pwd_context.verify(plain_password, hashed_password)


def create_access_token(data: dict, expires_delta: Optional[timedelta] = None) -> str:
    to_encode = data.copy()
    if expires_delta:
        expire = datetime.utcnow() + expires_delta
    else:
        expire = datetime.utcnow() + timedelta(minutes=settings.ACCESS_TOKEN_EXPIRE_MINUTES)
    to_encode.update({"exp": expire})
    encoded_jwt = jwt.encode(to_encode, settings.SECRET_KEY, algorithm=settings.ALGORITHM)
    return encoded_jwt


def decode_access_token(token: str) -> Optional[dict]:
    try:
        payload = jwt.decode(token, settings.SECRET_KEY, algorithms=[settings.ALGORITHM])
        return payload
    except JWTError:
        return None

class EmailService:
    def __init__(self):
        self.host = settings.SMTP_HOST
        self.port = settings.SMTP_PORT
        self.username = settings.SMTP_USER
        self.password = settings.SMTP_PASSWORD
        self.from_email = settings.SMTP_FROM
        self.frontend_url = settings.FRONTEND_URL

    async def send_email(
        self,
        to_email: str,
        subject: str,
        html_content: str,
        text_content: Optional[str] = None
    ) -> bool:
        """Send an email using SMTP."""
        if not self.username or not self.password:
            print(f"Email not configured. Would send to {to_email}: {subject}")
            return False

        message = MIMEMultipart("alternative")
        message["From"] = self.from_email
        message["To"] = to_email
        message["Subject"] = subject

        if text_content:
            message.attach(MIMEText(text_content, "plain"))
        message.attach(MIMEText(html_content, "html"))

        try:
            await aiosmtplib.send(
                message,
                hostname=self.host,
                port=self.port,
                username=self.username,
                password=self.password,
                start_tls=True
            )
            return True
        except Exception as e:
            print(f"Failed to send email: {e}")
            return False

from datetime import datetime, timedelta, date
from apscheduler.schedulers.asyncio import AsyncIOScheduler
from apscheduler.triggers.cron import CronTrigger
from sqlalchemy.orm import Session
from app.core.database import SessionLocal
from app.models.models import User, Project, TimeEntry
from app.services.email_service import email_service

scheduler = AsyncIOScheduler()


def get_week_date_range():
    """Get the start and end dates of the current week."""
    today = date.today()
    start = today - timedelta(days=today.weekday())  # Monday
    end = start + timedelta(days=6)  # Sunday
    return start, end


async def send_weekly_manager_reports():
    """Send weekly pending time entries report to all project managers."""
    db = SessionLocal()
    try:
        week_start, week_end = get_week_date_range()
        
        # Get all active projects with managers
        projects = db.query(Project).filter(Project.status == True).all()
        
        # Group entries by manager
        manager_entries = {}
        
        for project in projects:
            # Get pending entries for this project in the current week
            entries = db.query(TimeEntry).filter(
                TimeEntry.project_id == project.id,
                TimeEntry.status == "PENDING",
                TimeEntry.work_date >= week_start,
                TimeEntry.work_date <= week_end
            ).all()
            
            for entry in entries:
                user = db.query(User).filter(User.id == entry.user_id).first()
                
                entry_data = {
                    "user_name": user.name if user else "Unknown",
                    "project_name": project.name,
                    "work_date": entry.work_date.isoformat(),
                    "hours": float(entry.hours),
                    "notes": entry.notes or ""
                }
                
                if project.manager_id not in manager_entries:
                    manager = db.query(User).filter(User.id == project.manager_id).first()
                    manager_entries[project.manager_id] = {
                        "manager_email": manager.email if manager else "",
                        "manager_name": manager.name if manager else "",
                        "entries": []
                    }
                
                manager_entries[project.manager_id]["entries"].append(entry_data)
        
        # Send emails to each manager
        for manager_id, data in manager_entries.items():
            if data["manager_email"] and data["entries"]:
                await email_service.send_weekly_report(
                    manager_email=data["manager_email"],
                    manager_name=data["manager_name"],
                    pending_entries=data["entries"],
                    week_start=week_start.strftime("%Y-%m-%d"),
                    week_end=week_end.strftime("%Y-%m-%d")
                )
                
    finally:
        db.close()


def start_scheduler():
    """Start the APScheduler for scheduled tasks."""
    # Schedule weekly report every Sunday at 9 AM
    scheduler.add_job(
        send_weekly_manager_reports,
        CronTrigger(day_of_week="sun", hour=9, minute=0),
        id="weekly_manager_report",
        replace_existing=True
    )
    
    scheduler.start()
    print("Scheduler started. Weekly reports will be sent on Sundays at 9 AM.")


def stop_scheduler():
    """Stop the scheduler."""
    scheduler.shutdown()

import React, { useState, useEffect } from 'react';
import { BrowserRouter, Routes, Route, Navigate } from 'react-router-dom';
import useAuthStore from './stores/authStore';
import Login from './components/Login';
import Register from './components/Register';
import Dashboard from './components/Dashboard';
import Timesheets from './components/Timesheets';
import Approvals from './components/Approvals';
import Projects from './components/Projects';
import Reports from './components/Reports';
import UserProvisioning from './components/UserProvisioning';
import Layout from './components/Layout';

function App() {
  const { isAuthenticated, fetchCurrentUser, token } = useAuthStore();
  const [initializing, setInitializing] = useState(true);

  useEffect(() => {
    const init = async () => {
      if (token) {
        await fetchCurrentUser();
      }
      setInitializing(false);
    };
    init();
  }, [token, fetchCurrentUser]);

  if (initializing) {
    return (
      <div className="loading-container">
        <div className="spinner"></div>
      </div>
    );
  }

  return (
    <BrowserRouter>
      <Routes>
        <Route
          path="/"
          element={
            isAuthenticated ? (
              <Navigate to="/dashboard" replace />
            ) : (
              <Login />
            )
          }
        />
        <Route path="/register" element={<Register />} />
        <Route
          path="/*"
          element={
            isAuthenticated ? (
              <Layout>
                <Routes>
                  <Route path="/dashboard" element={<Dashboard />} />
                  <Route path="/timesheets" element={<Timesheets />} />
                  <Route path="/approvals" element={<Approvals />} />
                  <Route path="/projects" element={<Projects />} />
                  <Route path="/reports" element={<Reports />} />
                  <Route path="/users" element={<UserProvisioning />} />
                </Routes>
              </Layout>
            ) : (
              <Navigate to="/" replace />
            )
          }
        />
      </Routes>
    </BrowserRouter>
  );
}

export default App;

import {

} from 'lucide-react';

const Timesheets = () => {
  const [currentDate, setCurrentDate] = useState(new Date());
  const [entries, setEntries] = useState([]);
  const [projects, setProjects] = useState([]);
  const [loading, setLoading] = useState(true);
  const [showModal, setShowModal] = useState(false);
  const [selectedDate, setSelectedDate] = useState(null);
  const [formData, setFormData] = useState({
    project_id: '',
    hours: '',
    notes: '',
  });
  const [viewMode, setViewMode] = useState('calendar'); // calendar or list

  useEffect(() => {
    fetchProjects();
  }, []);

  useEffect(() => {
    fetchEntries();
  }, [currentDate]);

  const fetchProjects = async () => {
    try {
      const response = await projectsAPI.list();
      setProjects(response.data.data.projects || []);
    } catch (error) {
      console.error('Error fetching projects:', error);
    }
  };

  const fetchEntries = async () => {
    setLoading(true);
    try {
      const year = currentDate.getFullYear();
      const month = currentDate.getMonth();
      const startDate = new Date(year, month, 1);
      const endDate = new Date(year, month + 1, 0);

      const response = await timeEntriesAPI.list({
        start_date: startDate.toISOString().split('T')[0],
        end_date: endDate.toISOString().split('T')[0],
      });

      setEntries(response.data.data.entries || []);
    } catch (error) {
      console.error('Error fetching entries:', error);
    } finally {
      setLoading(false);
    }
  };

  const getDaysInMonth = (date) => {
    const year = date.getFullYear();
    const month = date.getMonth();
    const firstDay = new Date(year, month, 1);
    const lastDay = new Date(year, month + 1, 0);
    const daysInMonth = lastDay.getDate();
    const startingDay = firstDay.getDay();

    const days = [];
    for (let i = 0; i < startingDay; i++) {
      days.push({ date: null, isOtherMonth: true });
    }
    for (let i = 1; i <= daysInMonth; i++) {
      days.push({
        date: new Date(year, month, i),
        isOtherMonth: false,
      });
    }
    return days;
  };

  const getEntriesForDate = (date) => {
    if (!date) return [];

import axios from 'axios';

const API_BASE_URL = process.env.REACT_APP_API_URL || 'http://localhost:8000/api';

const api = axios.create({
  baseURL: API_BASE_URL,
  headers: {
    'Content-Type': 'application/json',
  },
});

// Request interceptor to add auth token
api.interceptors.request.use(
  (config) => {
    const token = localStorage.getItem('token');
    if (token) {
      config.headers.Authorization = `Bearer ${token}`;
    }
    return config;
  },
  (error) => {
    return Promise.reject(error);
  }
);

// Response interceptor to handle errors
api.interceptors.response.use(
  (response) => response,
  (error) => {
    if (error.response?.status === 401) {
      localStorage.removeItem('token');
      localStorage.removeItem('user');
      window.location.href = '/';
    }
    return Promise.reject(error);
  }
);

    set({ loading: true, error: null });
    try {
      const response = await authAPI.login(email, password);
      const { data } = response.data;
      
      if (response.data.status === 'error') {
        set({ loading: false, error: response.data.message });
        return false;
      }

      localStorage.setItem('token', data.token);
      localStorage.setItem('user', JSON.stringify(data.profile));
      
      set({
        user: data.profile,
        token: data.token,
        isAuthenticated: true,
        loading: false,
        error: null,
      });
      
      return true;
    } catch (error) {
      const message = error.response?.data?.message || 'Login failed';
      set({ loading: false, error: message });
      return false;
    }
  },

  register: async (userData) => {
    set({ loading: true, error: null });
    try {
      const response = await authAPI.register(userData);
      
      if (response.data.status === 'error') {
        set({ loading: false, error: response.data.message });
        return false;
      }

      set({ loading: false });
      return true;
    } catch (error) {
      const message = error.response?.data?.message || 'Registration failed';
      set({ loading: false, error: message });
      return false;
    }
  },

  logout: () => {
    localStorage.removeItem('token');
    localStorage.removeItem('user');
    set({
      user: null,
      token: null,
      isAuthenticated: false,
      error: null,
    });
  },

  fetchCurrentUser: async () => {
    try {
      const response = await authAPI.getCurrentUser();
      const user = response.data.data;
      localStorage.setItem('user', JSON.stringify(user));
      set({ user });
    } catch (error) {
      console.error('Failed to fetch current user:', error);
    }
  },

  hasRole: (roleName) => {
    const user = get().user;
    if (!user || !user.roles) return false;
    return user.roles.some(role => role.value === roleName);
  },

  isAdmin: () => get().hasRole('Admin'),
  isManager: () => get().hasRole('Manager'),
  isMember: () => get().hasRole('Member'),

  clearError: () => set({ error: null }),
}));

export default useAuthStore;

import { useAuthStore } from '@/stores/auth-store';

const API_BASE = "https://timetracker-api-m8k7.onrender.com/api";

export class ApiError extends Error {
  statusCode: number;
  constructor(message: string, statusCode: number) {
    super(message);
    this.statusCode = statusCode;
  }
}

async function request<T>(endpoint: string, options: RequestInit = {}): Promise<T> {
  const token = useAuthStore.getState().token;
  const headers: HeadersInit = { 'Content-Type': 'application/json', ...options.headers };
  if (token) { (headers as Record<string, string>)['Authorization'] = `Bearer ${token}`; }
  
  const url = `${API_BASE}${endpoint}`;
  const response = await fetch(url, { ...options, headers });
  const data = await response.json();

  if (!response.ok) {
    throw new ApiError(data.detail || data.message || 'An error occurred', response.status);
  }
  return data;
}

export const api = {
  get: <T>(endpoint: string) => request<T>(endpoint, { method: 'GET' }),
  post: <T>(endpoint: string, body?: unknown) => request<T>(endpoint, { method: 'POST', body: body ? JSON.stringify(body) : undefined }),
  put: <T>(endpoint: string, body?: unknown) => request<T>(endpoint, { method: 'PUT', body: body ? JSON.stringify(body) : undefined }),
  delete: <T>(endpoint: string) => request<T>(endpoint, { method: 'DELETE' }),
};

export const authApi = {
  login: async (email: string, password: string) => {
    const result = await api.post<{ access_token: string; user: any }>('/auth/login', { email, password });
    return {
      profile: {
        id: result.user.id,
        name: `${result.user.first_name} ${result.user.last_name}`,
        email: result.user.email,
        roles: result.user.roles.map((r: any) => ({ id: r.id, value: r.name }))
      },
      token: result.access_token
    };
  },
  me: async () => {
    const user = await api.get<any>('/auth/me');
    return {
      id: user.id,
      name: `${user.first_name} ${user.last_name}`,
      email: user.email,
      roles: user.roles.map((r: any) => ({ id: r.id, value: r.name }))
    };
  }
};

export const projectsApi = {
  list: async () => {
    const projects = await api.get<any[]>('/projects');
    return { projects: projects.map(p => ({ id: p.id, name: p.name, code: p.code, status: p.is_active })) };
  },
  create: (data: { name: string; code: string }) => api.post<any>('/projects', data)
};

export const timeEntriesApi = {
  list: (params?: any) => api.get<any[]>('/time-entries'),
  create: (data: any) => api.post<any>('/time-entries', data)
};
